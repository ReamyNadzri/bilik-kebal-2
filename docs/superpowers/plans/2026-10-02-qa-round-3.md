## Findings that shape the plan

1. **#1 root cause (confirmed on Cloud).** `202610160001_clerk_identity_bridge.sql` rewrote every `auth.uid()` to `private.current_user_id()` once. `202610170001_google_profile_photo.sql` came **after** it and re-introduced `auth.uid()` in two functions. Read-only query on Cloud: the only functions/policies still containing `auth.uid()` are `public.use_own_external_avatar` and `public.console_reset_member_avatar` (so the Sheriff "Reset avatar" is broken too).
2. **#4** Sessions are rows, not code. Cloud has `uitm / sem-1-2024-2025 "Semester 1, 2024/2025"` (used by 3 Wanteds) and `sem-2-2024-2025` (unused). Rename in place (ids kept, nothing re-pointed) and add Semesters 3–6.
3. **#5** The 20-character floor lives in five places: `wanted_requests_description_check`, `update_own_published_wanted`, `src/contracts/marketplace.ts` (×3), `src/features/marketplace/wanted-draft.ts` (`DESCRIPTION_MIN`), `wanted-poster-controls.tsx`, and the poster service messages.
4. **#7** Course code is a native `<input list>` + `<datalist>`; the browser draws the popup, so it cannot be styled or aligned. It must stay free-text (members may type a new code, `ensure_course` files it), so `SearchSelect` (choose-only) does not fit — a sibling free-text combobox does.
5. **#8** `contributions` already has `contributions_read_own` RLS (rewritten to `current_user_id()`), and `SupabaseWantedRepository.listWantedByIds` already builds `WantedSummary` cards. Backings stay private: `/profile` only, never `/u/[publicId]`.

```ts
/** The SQL with -- comments removed, so explanations may still name auth.uid(). */
function code(sql: string): string {
  return sql
    .split("\n")
    .map((line) => line.replace(/--.*$/, ""))
    .join("\n");
}

/** The last definition of each function wins, as it does in the database. */
function liveOffenders(): string[] {
  const files = readdirSync(DIR).filter((name) => name > BRIDGE && name.endsWith(".sql")).sort();
  const live = new Map<string, string>();
  for (const file of files) {
    const sql = code(readFileSync(join(DIR, file), "utf8"));
    const pattern = /create\s+(?:or\s+replace\s+)?function\s+([\w.]+)\s*\(([\s\S]*?)\$\$([\s\S]*?)\$\$/gi;

      live.set(match[1]!.toLowerCase(), `${file}:${match[3]!.includes("auth.uid()") ? "uid" : "ok"}`);
    }
  }
  return [...live].filter(([, state]) => state.endsWith(":uid")).map(([name, state]) => `${name} (${state.split(":")[0]})`);
}
```

```sql
-- brought auth.uid() back in two functions. A Clerk subject is `user_…`, so the uuid cast failed
-- and "Use my Google photo" and the Sheriff's avatar reset always answered "could not be saved".

create or replace function public.use_own_external_avatar()
returns text
language plpgsql
security definer
set search_path = ''
as $$
declare
  member_id uuid := private.current_user_id();
  avatar_url text;
begin
  if member_id is null then
    raise exception using errcode = '42501', message = 'NOT_AUTHORIZED';
  end if;

  select clerk_avatar_url into avatar_url
  from private.user_identities
  where user_id = member_id;

  if avatar_url is null then
    raise exception using errcode = '22023', message = 'external_avatar_missing';
  end if;

  update public.profiles
  set avatar_object_key = null,
      avatar_preset = null,
      external_avatar_enabled = true,
      external_avatar_url = avatar_url,
      updated_at = now()
  where user_id = member_id;

  return avatar_url;
end;
$$;

create or replace function public.console_reset_member_avatar(target_public_id uuid, reason_code text)
returns void
language plpgsql
security definer
set search_path = ''
as $$
declare
  member_id uuid;
begin
  perform private.console_write_guard(array['owner', 'platform_sheriff']);
  perform private.check_reason_code(reason_code);
  member_id := private.member_by_public_id(target_public_id);
  if exists (select 1 from public.platform_role_assignments where user_id = member_id and role = 'owner')
    and member_id <> private.current_user_id() then
    raise exception using errcode = '42501', message = 'NOT_AUTHORIZED';
  end if;
  update public.profiles
  set avatar_object_key = null,
      avatar_preset = null,
      external_avatar_enabled = false,
      external_avatar_url = null,
      updated_at = now()
  where user_id = member_id;
  insert into public.identity_audit_events (event_type, actor_user_id, subject_user_id, details)
  values ('profile.avatar_reset_by_staff', private.current_user_id(), member_id, jsonb_build_object('reason_code', reason_code));
end;
$$;
```

```tsx
"use client";

import { useEffect, useId, useState, type KeyboardEvent } from "react";
import type { CourseOption } from "@/contracts/marketplace";
import { normaliseCourseCode } from "@/features/marketplace/wanted-draft";
import { matchesQuery } from "./search-select";

interface CourseCodeComboboxProps {
  readonly id: string;
  readonly value: string;
  readonly courses: readonly CourseOption[];
  readonly error: string | undefined;
  readonly onChange: (code: string) => void;
}

const MAX_SHOWN = 8;

/**
 * The course code: free text (a new code is filed by the server), with the
 * known courses suggested in a list the app draws under the field. Same
 * keyboard as SearchSelect, but whatever is typed stays the value.
 */
export function CourseCodeCombobox({ id, value, courses, error, onChange }: CourseCodeComboboxProps) {
  const listId = useId();
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(-1);

  const matches = courses
    .filter((course) => value === "" || matchesQuery(`${course.code} ${course.label}`, value))
    .slice(0, MAX_SHOWN);
  const activeOption = open && active >= 0 ? matches[active] : undefined;
  const optionId = (course: CourseOption) => `${listId}-${course.id}`;
  const showList = open && matches.length > 0;

  useEffect(() => {
    if (activeOption) document.getElementById(optionId(activeOption))?.scrollIntoView?.({ block: "nearest" });
  });

  function choose(course: CourseOption) {
    onChange(course.code);
    setOpen(false);
    setActive(-1);
  }

  function onKeyDown(event: KeyboardEvent<HTMLInputElement>) {
    if (event.key === "ArrowDown" || event.key === "ArrowUp") {
      event.preventDefault();
      setOpen(true);
      const step = event.key === "ArrowDown" ? 1 : -1;
      setActive((current) => (current + step + matches.length) % Math.max(matches.length, 1));
    } else if (event.key === "Enter" && activeOption) {
      event.preventDefault();
      choose(activeOption);
    } else if (event.key === "Escape" && open) {
      event.preventDefault();
      setOpen(false);
      setActive(-1);
    }
  }

  const hintId = `${id}-hint`;
  const errorId = `${id}-error`;

  return (
    <div className="form-field search-select">
      <label className="form-field__label" htmlFor={id}>
        Course code<span className="form-field__required"> (required)</span>
      </label>
      <p className="form-field__hint" id={hintId}>
        Type the subject code in capital letters, for example CSC510.
      </p>
      {error === undefined ? null : (
        <p className="form-field__error" id={errorId}>
          {error}
        </p>
      )}
      <input
        className="form-field__input"
        id={id}
        type="text"
        role="combobox"
        value={value}
        maxLength={12}
        autoComplete="off"
        autoCapitalize="characters"
        spellCheck={false}
        aria-autocomplete="list"
        aria-expanded={showList}
        aria-controls={listId}
        aria-activedescendant={activeOption ? optionId(activeOption) : undefined}
        aria-invalid={error === undefined ? undefined : true}
        aria-describedby={error === undefined ? hintId : `${hintId} ${errorId}`}
        onFocus={() => setOpen(true)}
        onClick={() => setOpen(true)}
        onBlur={() => setOpen(false)}
        onKeyDown={onKeyDown}
        onChange={(event) => {
          onChange(normaliseCourseCode(event.target.value));
          setOpen(true);
          setActive(-1);
        }}
      />
      <ul className="search-select__list" id={listId} role="listbox" aria-label="Known courses" hidden={!showList}>
        {matches.map((course, index) => (
          <li
            key={course.id}
            id={optionId(course)}
            role="option"
            aria-selected={index === active}
            className="search-select__option course-option"
            // mousedown, not click: it lands before the input's blur closes the list.
            onMouseDown={(event) => {
              event.preventDefault();
              choose(course);
            }}
          >
            <span className="course-option__code">{course.code}</span>{" "}
            {course.label === course.code ? null : <span className="course-option__name">{course.label}</span>}
          </li>
        ))}
      </ul>
    </div>
  );
}
```

```ts
/**
 * Wanteds the member contributed to, excluding their own posts, most
 * recently backed first. RLS (`contributions_read_own`) limits the read to
 * the caller's own contributions.
 */
async listBackedWantedIds(memberUserId: string): Promise<string[]> {
  const contributions = await this.client
    .from("contributions")
    .select("wanted_request_id, created_at")
    .eq("contributor_user_id", memberUserId)
    .order("created_at", { ascending: false });
  if (contributions.error) throw contributions.error;
  const ids = [...new Set(contributions.data.map((row) => row.wanted_request_id))];
  if (ids.length === 0) return [];
  const own = await this.client
    .from("wanted_requests")
    .select("id")
    .in("id", ids)
    .eq("commissioner_user_id", memberUserId);
  if (own.error) throw own.error;
  const posted = new Set(own.data.map((row) => row.id));
  return ids.filter((id) => !posted.has(id));
}
```
